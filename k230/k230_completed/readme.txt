 模式切换及状态设置（0x00–0x0F）

  ┌───────┬──────┬─────────────┬───────────────────────────────────────┐
  │ ASCII │ 字节 │    宏名     │                 功能                  │
  ├───────┼──────┼─────────────┼───────────────────────────────────────┤
  │ S     │ 0x01 │ START       │ 开始检测                              │
  ├───────┼──────┼─────────────┼───────────────────────────────────────┤
  │ T     │ 0x02 │ STOP        │ 停止检测                              │
  ├───────┼──────┼─────────────┼───────────────────────────────────────┤
  │ M     │ 0x03 │ MUTE        │ 静默/恢复串口发送                     │
  ├───────┼──────┼─────────────┼───────────────────────────────────────┤
  │ C     │ 0x04 │ SWITCH_CROP │ 循环切换作物（玉米→甘蔗→番茄→玉米） │
  ├───────┼──────┼─────────────┼───────────────────────────────────────┤
  │ P     │ 0x05 │ MODE_PEST   │ 切换病虫害检测模式                    │
  ├───────┼──────┼─────────────┼───────────────────────────────────────┤
  │ R     │ 0x06 │ MODE_TRACK  │ 切换人体追踪模式                      │
  ├───────┼──────┼─────────────┼───────────────────────────────────────┤
  │ H     │ 0x07 │ SERVO_HOME  │ 舵机归中                              │
  ├───────┼──────┼─────────────┼───────────────────────────────────────┤
  │ A     │ 0x08 │ SERVO_LEFT  │ 舵机左转 +15°（仅 MANUAL）            │
  ├───────┼──────┼─────────────┼───────────────────────────────────────┤
  │ D     │ 0x09 │ SERVO_RIGHT │ 舵机右转 −15°（仅 MANUAL）            │
  ├───────┼──────┼─────────────┼───────────────────────────────────────┤
  │ Z     │ 0x0A │ SERVO_DOWN  │ 舵机下转 −15°（仅 MANUAL）            │
  ├───────┼──────┼─────────────┼───────────────────────────────────────┤
  │ W     │ 0x0B │ SERVO_UP    │ 舵机上转 +15°（仅 MANUAL）            │
  ├───────┼──────┼─────────────┼───────────────────────────────────────┤
  │ O     │ 0x0C │ TOGGLE_MODE │ 切换 MANUAL / AUTO 模式               │
  ├───────┼──────┼─────────────┼───────────────────────────────────────┤
  │ 1     │ 0x0D │ CROP_CORN   │ 直接切换玉米                          │
  ├───────┼──────┼─────────────┼───────────────────────────────────────┤
  │ 2     │ 0x0E │ CROP_POTATO │ 直接切换甘蔗                        │
  ├───────┼──────┼─────────────┼───────────────────────────────────────┤
  │ 3     │ 0x0F │ CROP_TOMATO │ 直接切换番茄                          │
  └───────┴──────┴─────────────┴───────────────────────────────────────┘

  ┌──────┬──────────┬──────────────────────────────┐
  │ 按键 │ 发送字节 │             功能             │
  ├──────┼──────────┼──────────────────────────────┤
  │ F    │ 0x10     │ 触发搜索（仅 AUTO 模式有效） │
  ├──────┼──────────┼──────────────────────────────┤
  │ X    │ 0x11     │ 强制中止搜索 / 归 IDLE       │
  └──────┴──────────┴──────────────────────────────┘
